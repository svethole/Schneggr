cmake_minimum_required(VERSION 3.21)
project(Schneggr VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)
qt_standard_project_setup()
qt_add_executable(Schneggr WIN32 src/main.cpp)
target_link_libraries(Schneggr PRIVATE Qt6::Widgets Qt6::Svg)
if(WIN32)
 target_link_libraries(Schneggr PRIVATE user32)
endif()
